    <script>
        // Check PIN functionality
        function checkPin() {
            const correctPin = "1234"; // 👈 Your custom 4-digit PIN
            const userInput = document.getElementById("pin-input").value;
            const errorMsg = document.getElementById("error-msg");
            const loginScreen = document.getElementById("login-screen");
            const surpriseContent = document.getElementById("surprise-content");

            if (userInput === correctPin) {
                loginScreen.style.display = "none";
                surpriseContent.style.display = "block";
                startFloatingFlowers(); 
            } else {
                errorMsg.innerText = "Incorrect PIN. Try again, love! 💕";
                document.getElementById("pin-input").value = "";
            }
        }

        // Create random moving flowers
        function startFloatingFlowers() {
            const flowerContainer = document.getElementById('flower-container');
            const flowerTypes = ['🌸', '🌹', '🌷', '🌻', '🌺', '💐'];
            
            setInterval(() => {
                const flower = document.createElement('div');
                flower.classList.add('floating-flower');
                flower.innerText = flowerTypes[Math.floor(Math.random() * flowerTypes.length)];
                flower.style.left = Math.random() * 100 + 'vw';
                flower.style.fontSize = Math.random() * 20 + 20 + 'px';
                flower.style.animationDuration = Math.random() * 4 + 6 + 's';
                
                flowerContainer.appendChild(flower);
                
                setTimeout(() => {
                    flower.remove();
                }, 8000);
            }, 600); 
        }

        // Allow pressing 'Enter' key to submit PIN
        document.getElementById("pin-input").addEventListener("keypress", function(event) {
            if (event.key === "Enter") {
                checkPin();
            }
        });
    </script>
